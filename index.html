Carrinho Seguidor de Linha

Segue linha preta ou branca com dois sensores infravermelhos, desvia de obstáculos com o sensor ultrassônico HC-SR04 e esterça com um servomotor.

Ligações

  • Ultrassom Trig / Echo7 / 8

#include <Servo.h> // Importa a biblioteca necessária para controlar servomotores

Servo servoDirecao; // Cria o objeto que vai controlar o servo responsável pela direção

// Definição dos pinos da Ponte H (controle do motor de tração)
const int pinoVelocidade = 3; // Pino PWM para controlar a velocidade do motor
const int pinoDirecaoA   = 4; // Pino de controle do sentido de giro A
const int pinoDirecaoB   = 5; // Pino de controle do sentido de giro B

// Definição dos pinos do Sensor Ultrassônico (HC-SR04)
const int pinoPulsoUltrassom    = 7; // Pino Trig: envia o pulso sonoro
const int pinoRecepcaoUltrassom = 8; // Pino Echo: recebe o eco do som refletido

// Definição dos pinos dos Sensores Infravermelhos (seguidores de linha)
const int sensorEsq = A0; // Sensor posicionado à esquerda
const int sensorDir = A1; // Sensor posicionado à direita

// Pino da chave ou botão que alterna entre pista preta e pista branca
const int pinoSeletorModo = 6;

void setup() {
  // Configura o pino do seletor usando o resistor pull-up interno do Arduino
  pinMode(pinoSeletorModo, INPUT_PULLUP);
  
  // Configura os pinos do sensor ultrassônico
  pinMode(pinoPulsoUltrassom, OUTPUT);
  pinMode(pinoRecepcaoUltrassom, INPUT);
  
  // Configura os pinos dos sensores de linha como entradas
  pinMode(sensorEsq, INPUT);
  pinMode(sensorDir, INPUT);
  
  // Configura os pinos da Ponte H como saídas
  pinMode(pinoVelocidade, OUTPUT);
  pinMode(pinoDirecaoA, OUTPUT);
  pinMode(pinoDirecaoB, OUTPUT);
  
  // Conecta o servomotor ao pino digital 9 e centraliza as rodas em 45 graus
  servoDirecao.attach(9);
  servoDirecao.write(45);
}

// Função responsável por calcular a distância até um obstáculo em centímetros
long lerDistancia() {
  // Garante que o pino Trig comece desligado
  digitalWrite(pinoPulsoUltrassom, LOW);
  delayMicroseconds(2);
  
  // Emite um pulso ultrassônico de 10 microssegundos
  digitalWrite(pinoPulsoUltrassom, HIGH);
  delayMicroseconds(10);
  digitalWrite(pinoPulsoUltrassom, LOW);
  
  // Mede o tempo (em microssegundos) que o som levou para ir e voltar
  long duracao = pulseIn(pinoRecepcaoUltrassom, HIGH);
  
  // Converte o tempo em distância (velocidade do som ~0.034 cm/µs, dividido por 2 pois é ida e volta)
  long distanciaCm = duracao * 0.034 / 2;
  return distanciaCm;
}

// Função auxiliar para acionar o motor para frente e esterçar o servo simultaneamente
void moverMotor(int velocidade, int anguloServo) {
  servoDirecao.write(anguloServo);     // Ajusta o ângulo da direção
  digitalWrite(pinoDirecaoA, HIGH);    // Define o sentido de rotação para frente
  digitalWrite(pinoDirecaoB, LOW);
  analogWrite(pinoVelocidade, velocidade); // Aplica a potência/velocidade via PWM (0 a 255)
}

// Função que executa uma manobra fixa para contornar um obstáculo à frente
void desviarObstaculo() {
  // Mantém o carrinho andando para frente em velocidade reduzida (75)
  digitalWrite(pinoDirecaoA, HIGH);
  digitalWrite(pinoDirecaoB, LOW);
  analogWrite(pinoVelocidade, 75);
  
  servoDirecao.write(0);   // Vira totalmente para a esquerda para sair da rota do obstáculo
  delay(1000);             // Avança virado por 1 segundo
  servoDirecao.write(45);  // Alinha as rodas para frente
  
  servoDirecao.write(120); // Vira para a direita para retornar ao trajeto
  delay(1000);             // Avança virado por 1 segundo
  servoDirecao.write(45);  // Centraliza novamente as rodas
}

void loop() {
  // Lê o estado atual dos dois sensores de linha (HIGH ou LOW) e a distância frontal
  int esq = digitalRead(sensorEsq);
  int dir = digitalRead(sensorDir);
  long distancia = lerDistancia();
  
  // 1ª Prioridade: Se detectar qualquer obstáculo válido a menos de 20 cm, inicia o desvio
  if (distancia > 0 && distancia < 20) {
    desviarObstaculo();
    return; // Interrompe o loop atual para reavaliar os sensores logo após a manobra
  }
  
  // Verifica se o botão/chave seletora está acionado (nível LOW por conta do INPUT_PULLUP)
  bool modoInvertido = (digitalRead(pinoSeletorModo) == LOW);

  if (modoInvertido) {
    // MODO INVERTIDO: Segue linha branca sobre fundo escuro (sensores retornam LOW na linha)
    if (esq == LOW && dir == LOW) {
      moverMotor(85, 45);  // Ambos na linha: segue reto em velocidade padrão (85)
    } else if (esq == LOW && dir == HIGH) {
      moverMotor(75, 120); // Saiu pela esquerda: reduz velocidade (75) e vira para a direita
    } else if (esq == HIGH && dir == LOW) {
      moverMotor(75, 0);   // Saiu pela direita: reduz velocidade (75) e vira para a esquerda
    } else {
      moverMotor(85, 45);  // Ambos fora da linha (HIGH): mantém trajetória reta
    }
  } else {
    // MODO PADRÃO: Segue linha preta sobre fundo claro (sensores retornam HIGH na linha)
    if (esq == HIGH && dir == HIGH) {
      moverMotor(85, 45);  // Ambos na linha: segue reto em velocidade padrão (85)
    } else if (esq == HIGH && dir == LOW) {
      moverMotor(75, 120); // Saiu pela esquerda: reduz velocidade (75) e vira para a direita
    } else if (esq == LOW && dir == HIGH) {
      moverMotor(75, 0);   // Saiu pela direita: reduz velocidade (75) e vira para a esquerda
    } else {
      moverMotor(85, 45);  // Ambos fora da linha (LOW): mantém trajetória reta
    }
  }
}
